const WEEKDAYS = ['mon','tue','wed','thu','fri'];

function uid(){ return Math.random().toString(36).slice(2,10); }

function pad(n){ return n<10 ? '0'+n : ''+n; }

function toISO(d){ return d.getFullYear()+'-'+pad(d.getMonth()+1)+'-'+pad(d.getDate()); }

function generateSchedule(startDate, numWeeks) {

  let counts = {}; families.forEach(f=>counts[f.id]=0);

  let lastIdx = {}; let idx = 0;

  Object.keys(schedule).sort().forEach(dateKey=>{

    ['AM','PM'].forEach(shift=>{

      const fid = schedule[dateKey][shift];

      if (fid && counts[fid]!==undefined) { counts[fid]++; lastIdx[fid]=idx; }

      idx++;

    });

  });

  let current = new Date(startDate);

  while (!isWeekday(current)) current = addDays(current,1);

  let filled = 0;

  const targetSlots = numWeeks * 5;

  while (filled < targetSlots) {

    const dateKey = toISO(current);

    const wkey = DAY_NUM_TO_KEY[current.getDay()];

    if (!schedule[dateKey]) schedule[dateKey] = {AM:null, PM:null};

    ['AM','PM'].forEach(shift=>{

      idx++;

      if (schedule[dateKey][shift]) return;

      let candidates = families.filter(f => availability[f.id]?.[wkey]?.[shift] !== false);

      if (candidates.length === 0) { schedule[dateKey][shift] = null; return; }

      candidates.sort((a,b)=>{

        const ca = counts[a.id]||0, cb = counts[b.id]||0;

        if (ca !== cb) return ca - cb;

        const la = lastIdx[a.id] ?? -1, lb = lastIdx[b.id] ?? -1;

        return la - lb;

      });

      const chosen = candidates[0];

      schedule[dateKey][shift] = chosen.id;

      counts[chosen.id] = (counts[chosen.id]||0) + 1;

      lastIdx[chosen.id] = idx;

    });
